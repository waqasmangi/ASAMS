            showLoading();

        const listenForOwners = () => {
            if (!db) return;
            // Fetch all users marked as isOwner=true (requires security rules permitting this)
            const q = collection(db, `/artifacts/${appId}/users`); // Simplified path for demo
            onSnapshot(q, async (querySnapshot) => {
                const ownersList = document.getElementById('owners-list');
                if (!ownersList) return;
                ownersList.innerHTML = '';
                const owners = [];

                for (const docSnap of querySnapshot.docs) {
                    const detailsRef = doc(docSnap.ref, 'profile', 'details');
                    const detailsSnap = await getDoc(detailsRef);
                    if (detailsSnap.exists()) {
                        const ownerData = detailsSnap.data();
                        if (ownerData.isOwner) {
                            owners.push(ownerData);
                        }
                    }
                }

                if (owners.length === 0) {
                    ownersList.innerHTML = `<p class="p-4 bg-yellow-100 text-yellow-700 rounded-lg">No saloon owners registered yet.</p>`;
                    return;
                }

                owners.forEach(owner => {
                    const isValid = isTrialValid(owner.trialEnd);
                    const statusClass = isValid ? 'bg-green-100 text-green-700' : 'bg-red-100 text-red-700';
                    const statusText = isValid ? 'Trial Active / Paid' : 'Access Restricted (Trial Expired)';
                    ownersList.innerHTML += `
                        <div class="p-4 border rounded-lg shadow-sm flex justify-between items-center">
                            <div>
                                <p class="font-bold text-primary">${owner.name} (${owner.email})</p>

        const listenForOwnerData = (ownerId) => {

            e.preventDefault();
            if (!currentShopLocation) {

            showLoading();
            const shopName = e.target['shop-name'].value;
            const address = e.target['shop-address'].value;
            const phone = e.target['shop-phone'].value;

            try {
                const shopRef = doc(db, `/artifacts/${appId}/public/data/shops`, user.uid);
                await setDoc(shopRef, {
                    ownerId: user.uid,
                    name: shopName,
                    address,
                    phone,
                    location: currentShopLocation,
                    services: currentShopData?.services || [],
                    // Store owner profile details for easy customer viewing
                    ownerName: userData.name,
                    ownerEmail: userData.email,

            } catch (error) {
                console.error("Shop Setup Error:", error);
                alert(`Failed to save shop profile: ${error.message}`);
            } finally {
                hideLoading();
            }

        const handleOwnerAddService = async (e) => {
            e.preventDefault();
            if (!currentShopData) {
                alert("Please save your shop profile first.");
                return;
            }
            showLoading();
            const serviceName = e.target['service-name'].value;
            const price = parseInt(e.target['service-price'].value);
            const discount = parseInt(e.target['service-discount'].value || 0);

            const newService = { name: serviceName, price, discount };
            const updatedServices = [...(currentShopData.services || []), newService];

            try {
                const shopRef = doc(db, `/artifacts/${appId}/public/data/shops`, user.uid);
                await updateDoc(shopRef, { services: updatedServices });
                e.target.reset(); // Clear form
            } catch (error) {
                console.error("Add Service Error:", error);
                alert(`Failed to add service: ${error.message}`);
            } finally {
                hideLoading();
            }
        };

        const handleOwnerDeleteService = async (e) => {
            const index = parseInt(e.currentTarget.dataset.index);
            if (!currentShopData || index === undefined) return;
            showLoading();

            try {
                const updatedServices = currentShopData.services.filter((_, i) => i !== index);
                const shopRef = doc(db, `/artifacts/${appId}/public/data/shops`, user.uid);
                await updateDoc(shopRef, { services: updatedServices });
            } catch (error) {
                console.error("Delete Service Error:", error);
                alert(`Failed to delete service: ${error.message}`);
            } finally {
                hideLoading();
            }
        };

        const listenForAppointments = (ownerId) => {
            if (!db) return;
            const q = query(
                collection(db, `/artifacts/${appId}/public/data/appointments`),
                where("ownerId", "==", ownerId)
                // We'll sort in memory since orderBy can require complex indexes
            );

            onSnapshot(q, (querySnapshot) => {
                const appointmentsList = document.getElementById('appointments-list');
                if (!appointmentsList) return;
                appointmentsList.innerHTML = '';
                let appointments = [];
                querySnapshot.forEach(doc => {
                    appointments.push({ id: doc.id, ...doc.data() });
                });

                // Sort appointments by date
                appointments.sort((a, b) => (a.timestamp - b.timestamp));

                if (appointments.length === 0) {
                    appointmentsList.innerHTML = `<p class="text-accent">No upcoming appointments.</p>`;
                    return;
                }

                appointments.forEach(app => {
                    const date = new Date(app.timestamp);
                    const statusClass = app.status === 'Confirmed' ? 'bg-green-100 text-green-700' : 'bg-yellow-100 text-yellow-700';
                    appointmentsList.innerHTML += `
                        <div class="p-4 border rounded-lg shadow-sm bg-white">
                            <div class="flex justify-between items-start">
                                <p class="font-bold text-primary">${app.serviceName} (${app.stylist})</p>
                                <span class="text-xs font-semibold py-1 px-3 rounded-full ${statusClass}">${app.status}</span>
                            </div>
                            <p class="text-sm text-secondary">Customer: ${app.customerName} (${app.customerPhone})</p>
                            <p class="text-sm text-secondary">Date: ${date.toLocaleDateString()} at ${date.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })}</p>
                            <p class="text-xs text-accent mt-2">Appointment ID: ${app.id.substring(0, 8)}</p>
                        </div>
                    `;
                });
            });

        // --- Customer Module ---

        let allShops = [];
        let selectedShop = null;
        let selectedService = null;
        let selectedDate = new Date();
        let selectedStylist = "Any Available";

        const renderCustomer = () => {
            document.getElementById('app').innerHTML = `

                <div class="grid grid-cols-1 lg:grid-cols-3 gap-8">

                            <p class="text-accent">Click 'Get My Location' to find nearby saloons.</p>

                    <!-- Services & Booking (Matching UI Design - Booking Screen) -->
                    <div class="card lg:col-span-2">
                        <h2 class="text-xl font-bold text-primary border-b pb-3 mb-4">2. Book Appointment</h2>
                        ${selectedShop ? `<p class="text-success font-semibold mb-3">Selected Saloon: ${selectedShop.name}</p>` : `<p class="text-red-500 font-semibold mb-3">Please select a saloon from the left.</p>`}
                        
                        <div id="booking-container" class="${selectedShop ? '' : 'opacity-50 pointer-events-none'}">
                            <!-- Service Selection -->
                            <h3 class="font-bold text-secondary mt-4 mb-2">Available Services</h3>
                            <div id="customer-services" class="flex flex-wrap gap-2 mb-4">
                                ${selectedShop ? selectedShop.services.map(s => `<button data-service='${JSON.stringify(s)}' class="service-btn bg-muted text-primary text-sm py-1 px-3 rounded-full hover:bg-accent hover:text-white transition-colors">${s.name}</button>`).join('') : '<p class="text-sm text-accent">Select a saloon to see services.</p>'}
                            </div>

                            <!-- Calendar (Simplified to mimic the mobile view) -->
                            <h3 class="font-bold text-secondary mb-2">Select Date</h3>
                            <div class="flex justify-between p-3 bg-muted rounded-lg mb-4">
                                <button id="prev-day-btn" class="text-primary font-bold">&lt;</button>
                                <span id="selected-date-display" class="font-bold text-primary">${selectedDate.toLocaleDateString()}</span>
                                <button id="next-day-btn" class="text-primary font-bold">&gt;</button>
                            </div>

                            <!-- Stylist & Time (Simplified) -->
                            <h3 class="font-bold text-secondary mb-2">Select Stylist & Time</h3>
                            <div class="grid grid-cols-2 gap-4 mb-6">
                                <select id="stylist-select" class="input-field">
                                    <option value="Any Available">Any Available</option>
                                    <option value="Stylist Sarah">Stylist Sarah</option>
                                    <option value="Stylist Mike">Stylist Mike</option>
                                </select>
                                <select id="time-select" class="input-field">
                                    <option value="10:00 AM">10:00 AM</option>
                                    <option value="12:00 PM">12:00 PM</option>
                                    <option value="02:00 PM">02:00 PM</option>
                                    <option value="04:00 PM">04:00 PM</option>
                                </select>
                            </div>

                            <!-- Confirmation Area -->
                            <div class="bg-primary text-white p-4 rounded-xl space-y-2">
                                <p class="text-sm font-semibold">Selected Service: <span id="confirm-service-name">${selectedService?.name || 'N/A'}</span></p>
                                <p class="text-sm">Price: <span id="confirm-service-price">Rs. ${selectedService ? (selectedService.price * (1 - selectedService.discount / 100)).toFixed(0) : 'N/A'}</span></p>
                                <p class="text-sm">Stylist: <span id="confirm-stylist-name">${selectedStylist}</span></p>
                                <p class="text-sm">Time: <span id="confirm-time">10:00 AM</span></p>
                            </div>

                            <button id="confirm-booking-btn" class="btn-success w-full mt-6 py-3">CONFIRM BOOKING</button>
                        </div>
                    </div>
                </div>

            document.getElementById('get-location-customer-btn').addEventListener('click', getCustomerLocation);

            if (selectedShop) {
                document.querySelectorAll('.service-btn').forEach(btn => {
                    btn.addEventListener('click', handleServiceSelect);

                document.getElementById('prev-day-btn').addEventListener('click', () => changeSelectedDate(-1));
                document.getElementById('next-day-btn').addEventListener('click', () => changeSelectedDate(1));
                document.getElementById('stylist-select').addEventListener('change', (e) => {
                    selectedStylist = e.target.value;
                    document.getElementById('confirm-stylist-name').textContent = selectedStylist;
                });
                document.getElementById('time-select').addEventListener('change', (e) => {
                    document.getElementById('confirm-time').textContent = e.target.value;
                });
                document.getElementById('confirm-booking-btn').addEventListener('click', handleConfirmBooking);
            }

            // Initial render state for services/confirm
            if (selectedService) {
                document.getElementById('confirm-service-name').textContent = selectedService.name;
                document.getElementById('confirm-service-price').textContent = `Rs. ${(selectedService.price * (1 - selectedService.discount / 100)).toFixed(0)}`;
            }
        };

        const handleServiceSelect = (e) => {
            document.querySelectorAll('.service-btn').forEach(btn => btn.classList.remove('bg-accent', 'text-white'));
            e.currentTarget.classList.add('bg-accent', 'text-white');
            selectedService = JSON.parse(e.currentTarget.dataset.service);

            document.getElementById('confirm-service-name').textContent = selectedService.name;
            document.getElementById('confirm-service-price').textContent = `Rs. ${(selectedService.price * (1 - selectedService.discount / 100)).toFixed(0)}`;
        };

        const changeSelectedDate = (days) => {
            selectedDate.setDate(selectedDate.getDate() + days);
            document.getElementById('selected-date-display').textContent = selectedDate.toLocaleDateString();

        let customerLocation = null;

        const getCustomerLocation = () => {
            if (navigator.geolocation) {
                showLoading();
                navigator.geolocation.getCurrentPosition(
                    (position) => {
                        hideLoading();
                        customerLocation = {
                            lat: position.coords.latitude,
                            lng: position.coords.longitude
                        };
                        document.getElementById('customer-location-display').innerHTML = `Location: <span class="font-semibold">${customerLocation.lat.toFixed(4)}, ${customerLocation.lng.toFixed(4)}</span>`;
                        listenForShops(); // Start fetching shops once location is available
                    },
                    (error) => {
                        hideLoading();
                        console.error("Geolocation Error:", error);
                        alert("Could not get GPS location. Showing all shops.");
                        listenForShops(); // Show all shops as fallback
                    }
                );
            } else {
                alert("Geolocation is not supported by your browser. Showing all shops.");
                listenForShops(); // Show all shops as fallback
            }
        };

        const listenForShops = () => {
            if (!db) return;
            const q = collection(db, `/artifacts/${appId}/public/data/shops`);

            onSnapshot(q, (querySnapshot) => {
                const shopsList = document.getElementById('shops-list');
                if (!shopsList) return;
                shopsList.innerHTML = '';
                allShops = [];
                querySnapshot.forEach(doc => {
                    const shop = { id: doc.id, ...doc.data() };
                    // For a real app, calculate distance here using customerLocation and filter
                    allShops.push(shop);

                if (allShops.length === 0) {

                    return;
                }

                allShops.forEach(shop => {
                    // Simple distance simulation
                    const distanceText = customerLocation ? 'Nearby (Simulated)' : 'Global';
                    shopsList.innerHTML += `
                        <div data-shop='${JSON.stringify(shop)}' class="shop-item card p-4 border cursor-pointer hover:bg-muted">
                            <p class="font-bold text-primary text-lg">${shop.name}</p>
                            <p class="text-sm text-secondary">${shop.address}</p>
                            <p class="text-xs text-green-600">${distanceText}</p>
                        </div>
                    `;
                });

                document.querySelectorAll('.shop-item').forEach(item => {
                    item.addEventListener('click', handleShopSelect);
                });
            });
        };

        const handleShopSelect = (e) => {
            document.querySelectorAll('.shop-item').forEach(item => item.classList.remove('border-4', 'border-primary'));
            e.currentTarget.classList.add('border-4', 'border-primary');
            selectedShop = JSON.parse(e.currentTarget.dataset.shop);
            selectedService = selectedShop.services?.[0] || null; // Auto-select first service
            selectedStylist = "Any Available";
            renderCustomer(); // Re-render the booking panel with selected shop data
        };

        const handleConfirmBooking = async () => {
            if (!selectedShop || !selectedService) {
                alert("Please select a saloon and a service first.");
                return;
            }
            showLoading();

            const timeStr = document.getElementById('time-select').value;
            // Parse date and time into a single timestamp
            const appointmentTime = new Date(selectedDate.toDateString() + ' ' + timeStr);

            try {
                // 1. Save Appointment
                const appointmentData = {
                    ownerId: selectedShop.id,
                    ownerName: selectedShop.name,
                    customerUid: user.uid,
                    customerName: userData.name || 'Guest',
                    customerPhone: userData.phone || 'N/A', // Assuming phone is collected at signup in a real app
                    serviceName: selectedService.name,
                    stylist: selectedStylist,
                    price: selectedService.price,
                    discount: selectedService.discount,
                    timestamp: appointmentTime.getTime(),
                    status: 'Confirmed',
                    bookedAt: Date.now()
                };

                const appointmentsRef = collection(db, `/artifacts/${appId}/public/data/appointments`);
                const docRef = await addDoc(appointmentsRef, appointmentData);

                // 2. Send SMS Notifications (Simulated)

                // SMS to Customer
                await simulateTwilioSMS(
                    userData.phone || '+1234567890', // Placeholder phone
                    `Appointment confirmed at ${selectedShop.name} for ${selectedService.name} on ${appointmentTime.toLocaleDateString()} at ${timeStr}. Ref: ${docRef.id.substring(0, 5)}`
                );

                // SMS to Owner
                // NOTE: selectedShop has the owner's phone saved from the setup form
                await simulateTwilioSMS(
                    selectedShop.phone,
                    `NEW BOOKING at ${selectedShop.name}! Customer ${userData.name} for ${selectedService.name} on ${appointmentTime.toLocaleDateString()} at ${timeStr}.`
                );

                alert("Appointment successfully booked! Confirmation SMS simulated to you and the saloon owner.");
                window.location.reload(); // Reload customer view to clear form
            } catch (error) {
                console.error("Booking Error:", error);
                alert(`Failed to book appointment: ${error.message}`);
            } finally {
                hideLoading();
            }
        };

        // --- Main Router ---

        const router = () => {
            if (!isAuthReady) return; // Wait for Firebase Auth to initialize

            const hash = window.location.hash.substring(1);

            // 1. Initial Auth Check
            if (!user) {
                renderAuth();
                return;
            }

            // 2. Role-Based Routing
            if (hash === 'admin' && userData.isMasterAdmin) {
                renderAdmin();
            } else if (hash === 'owner' && userData.isOwner) {
                renderOwner();
            } else if (hash === 'customer' || !hash) {
                // Default view for customers/general users
                renderCustomer();
            } else {
                // User is logged in but trying to access an unauthorized route
                document.getElementById('app').innerHTML = `<div class="p-4 card mt-12 bg-red-100 text-red-700">
                    <h2 class="font-bold">Access Denied / Invalid Role</h2>
                    <p>You are logged in, but not authorized to access this page (${hash}).</p>
                    <div class="mt-4 space-y-2">
                        <button onclick="window.location.hash='#customer'" class="btn-primary bg-primary">Go to Customer Portal</button>
                        ${userData.isOwner ? `<button onclick="window.location.hash='#owner'" class="btn-primary bg-secondary">Go to Owner Portal</button>` : ''}
                        ${userData.isMasterAdmin ? `<button onclick="window.location.hash='#admin'" class="btn-primary bg-accent">Go to Admin Portal</button>` : ''}
                    </div>
                </div>`;
            }
        };

        // Initialize and listen for route changes
        initFirebaseAndAuth();
        window.addEventListener('hashchange', router);

        // --- Master Admin Setup Utility ---
        // This function should be run once by a real admin to set their own role.
        // For this demo, we'll hardcode the first registered user to be the admin for testing.
        onAuthStateChanged(auth, async (currentUser) => {
            if (currentUser && !currentUser.isAnonymous) {
                // Replace 'admin@example.com' with your actual admin email for testing
                if (currentUser.email === 'admin@example.com' || currentUser.uid === 'T9l5yO8qZlV0xW1eC4fPjA7bY2gD6sR3') {
                    // Update the user data if they match the designated admin UID/Email
                    const userRef = doc(db, `/artifacts/${appId}/users/${currentUser.uid}/profile`, 'details');
                    await setDoc(userRef, { isMasterAdmin: true, isOwner: false, name: 'Master Admin' }, { merge: true });
                }
            }
        });
